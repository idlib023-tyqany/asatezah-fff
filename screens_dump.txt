import 'package:flutter/material.dart';
import '../../services/admin_service.dart';

class AdminCatalog extends StatefulWidget {
  const AdminCatalog({super.key});

  @override
  State<AdminCatalog> createState() => _AdminCatalogState();
}

class _AdminCatalogState extends State<AdminCatalog> {
  final _admin = AdminService();
  List<dynamic> _categories = [];
  List<dynamic> _materials = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    setState(() => _loading = true);
    try {
      final c = await _admin.fetchCategories();
      final m = await _admin.fetchMaterials();
      setState(() {
        _categories = c;
        _materials = m;
        _loading = false;
      });
    } catch (_) {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _addKind(bool isCategory) async {
    final controller = TextEditingController();
    final name = await showDialog<String>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text(isCategory ? 'إضافة تصنيف' : 'إضافة مادة'),
        content: TextField(controller: controller, decoration: const InputDecoration(hintText: 'الاسم')),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('إلغاء')),
          FilledButton(onPressed: () => Navigator.pop(ctx, controller.text), child: const Text('إضافة')),
        ],
      ),
    );
    if (name == null || name.trim().isEmpty) return;
    if (isCategory) {
      await _admin.createCategory(name.trim());
    } else {
      await _admin.createMaterial(name.trim());
    }
    _load();
  }

  Future<void> _editKind(bool isCategory, dynamic item) async {
    final controller = TextEditingController(text: item['name'] ?? '');
    final name = await showDialog<String>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('تعديل'),
        content: TextField(controller: controller),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx), child: const Text('إلغاء')),
          FilledButton(onPressed: () => Navigator.pop(ctx, controller.text), child: const Text('حفظ')),
        ],
      ),
    );
    if (name == null || name.trim().isEmpty) return;
    if (isCategory) {
      await _admin.updateCategory(item['id'], name.trim());
    } else {
      await _admin.updateMaterial(item['id'], name.trim());
    }
    _load();
  }

  Future<void> _deleteKind(bool isCategory, dynamic item) async {
    if (isCategory) {
      await _admin.deleteCategory(item['id']);
    } else {
      await _admin.deleteMaterial(item['id']);
    }
    _load();
  }

  @override
  Widget build(BuildContext context) {
    return DefaultTabController(
      length: 2,
      child: Scaffold(
        appBar: AppBar(
          title: const Text('التصنيفات والمواد'),
          bottom: const TabBar(tabs: [Tab(text: 'التخصص / الصف'), Tab(text: 'المواد')]),
        ),
        body: _loading
            ? const Center(child: CircularProgressIndicator())
            : TabBarView(
                children: [
                  _list(_categories, isCategory: true),
                  _list(_materials, isCategory: false),
                ],
              ),
        floatingActionButton: FloatingActionButton(
          onPressed: () async {
            await showDialog<void>(
              context: context,
              builder: (ctx) => SimpleDialog(
                title: const Text('إضافة'),
                children: [
                  SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _addKind(true); }, child: const Text('تصنيف')),
                  SimpleDialogOption(onPressed: () { Navigator.pop(ctx); _addKind(false); }, child: const Text('مادة')),
                ],
              ),
            );
          },
          child: const Icon(Icons.add),
        ),
      ),
    );
  }

  Widget _list(List<dynamic> items, {required bool isCategory}) {
    if (items.isEmpty) {
      return const Center(child: Text('لا توجد عناصر', style: TextStyle(color: Colors.grey)));
    }
    return ListView.separated(
      itemCount: items.length,
      separatorBuilder: (_, _) => const Divider(height: 1),
      itemBuilder: (context, index) {
        final item = items[index];
        return ListTile(
          leading: Icon(isCategory ? Icons.category : Icons.menu_book),
          title: Text(item['name'] ?? ''),
          trailing: Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              IconButton(icon: const Icon(Icons.edit), onPressed: () => _editKind(isCategory, item)),
              IconButton(icon: const Icon(Icons.delete_outline, color: Colors.red), onPressed: () => _deleteKind(isCategory, item)),
            ],
          ),
        );
      },
    );
  }
}

import 'package:flutter/material.dart';
import '../../services/admin_service.dart';

class AdminFileReview extends StatefulWidget {
  final String status;
  final String title;

  const AdminFileReview({super.key, required this.status, required this.title});

  @override
  State<AdminFileReview> createState() => _AdminFileReviewState();
}

class _AdminFileReviewState extends State<AdminFileReview> {
  final _admin = AdminService();
  List<dynamic> _files = [];
  bool _loading = true;
  bool _isTrash = false;

  @override
  void initState() {
    super.initState();
    _isTrash = widget.status == 'trashed';
    _load();
  }

  Future<void> _load() async {
    setState(() => _loading = true);
    try {
      final result = await _admin.fetchFiles(status: widget.status);
      setState(() {
        _files = result['data'] as List<dynamic>? ?? [];
        _loading = false;
      });
    } catch (_) {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _action(int id, Future<void> Function() fn) async {
    try {
      await fn();
      _load();
    } catch (_) {
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('تعذّرت العملية')));
    }
  }

  Future<void> _moveDialog(Map<String, dynamic> file) async {
    final categories = await _admin.fetchCategories();
    final materials = await _admin.fetchMaterials();
    int? categoryId = file['category']?['id'];
    int? materialId = file['material']?['id'];
    if (!mounted) return;
    final newCategory = await showDialog<int>(
      context: context,
      builder: (ctx) => SimpleDialog(
        title: const Text('اختر التصنيف'),
        children: [
          for (final c in categories)
            SimpleDialogOption(
              onPressed: () => Navigator.pop(ctx, c['id']),
              child: Text(c['name'] ?? ''),
            ),
        ],
      ),
    );
    if (newCategory == null) return;
    categoryId = newCategory;
    if (!mounted) return;
    final newMaterial = await showDialog<int>(
      context: context,
      builder: (ctx) => SimpleDialog(
        title: const Text('اختر المادة'),
        children: [
          for (final m in materials)
            SimpleDialogOption(
              onPressed: () => Navigator.pop(ctx, m['id']),
              child: Text(m['name'] ?? ''),
            ),
        ],
      ),
    );
    if (newMaterial == null) return;
    materialId = newMaterial;
    await _action(file['id'], () => _admin.moveFile(file['id'], categoryId: categoryId, materialId: materialId));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text(widget.title)),
      body: _loading
          ? const Center(child: CircularProgressIndicator())
          : RefreshIndicator(
              onRefresh: _load,
              child: _files.isEmpty
                  ? ListView(
                      physics: const AlwaysScrollableScrollPhysics(),
                      children: const [SizedBox(height: 120), Center(child: Text('لا توجد ملفات', style: TextStyle(color: Colors.grey)))],
                    )
                  : ListView.separated(
                      itemCount: _files.length,
                      separatorBuilder: (_, _) => const Divider(height: 1),
                      itemBuilder: (context, index) {
                        final f = _files[index];
                        return _tile(f);
                      },
                    ),
            ),
    );
  }

  Widget _tile(Map<String, dynamic> f) {
    final uploader = f['uploader'];
    return ListTile(
      title: Text(f['title'] ?? '', maxLines: 1, overflow: TextOverflow.ellipsis),
      subtitle: Text(
        [
          f['category']?['name'] ?? '',
          f['material']?['name'] ?? '',
          uploader != null ? 'بواسطة: ${uploader['name'] ?? ''}' : '',
          _isTrash ? 'محذوف' : f['status'] ?? '',
        ].where((s) => s.isNotEmpty).join(' • '),
      ),
      trailing: _isTrash ? _trashActions(f) : _reviewActions(f),
    );
  }

  Widget _reviewActions(Map<String, dynamic> f) {
    final status = f['status'] ?? '';
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        if (status == 'pending')
          IconButton(
            icon: const Icon(Icons.check_circle, color: Colors.green),
            onPressed: () => _action(f['id'], () => _admin.approveFile(f['id'])),
          ),
        if (status == 'pending')
          IconButton(
            icon: const Icon(Icons.cancel, color: Colors.red),
            onPressed: () => _action(f['id'], () => _admin.rejectFile(f['id'])),
          ),
        if (status == 'rejected')
          IconButton(
            icon: const Icon(Icons.replay, color: Colors.orange),
            onPressed: () => _action(f['id'], () => _admin.reopenFile(f['id'])),
          ),
        if (status == 'published')
          IconButton(
            icon: const Icon(Icons.folder_open, color: Colors.blue),
            onPressed: () => _moveDialog(f),
          ),
        IconButton(
          icon: const Icon(Icons.delete_outline, color: Colors.red),
          onPressed: () => _action(f['id'], () => _admin.trashFile(f['id'])),
        ),
      ],
    );
  }

  Widget _trashActions(Map<String, dynamic> f) {
    return Row(
      mainAxisSize: MainAxisSize.min,
      children: [
        IconButton(
          icon: const Icon(Icons.restore, color: Colors.green),
          onPressed: () => _action(f['id'], () => _admin.restoreFile(f['id'])),
        ),
        IconButton(
          icon: const Icon(Icons.delete_forever, color: Colors.red),
          onPressed: () => _action(f['id'], () => _admin.forceDeleteFile(f['id'])),
        ),
      ],
    );
  }
}

import 'package:flutter/material.dart';
import '../../services/admin_service.dart';

class AdminNotify extends StatefulWidget {
  const AdminNotify({super.key});

  @override
  State<AdminNotify> createState() => _AdminNotifyState();
}

class _AdminNotifyState extends State<AdminNotify> {
  final _admin = AdminService();
  final _title = TextEditingController();
  final _body = TextEditingController();
  final _specialization = TextEditingController();
  bool _toAll = true;
  bool _sending = false;

  Future<void> _send() async {
    final title = _title.text.trim();
    final body = _body.text.trim();
    if (title.isEmpty || body.isEmpty) {
      _notify('أدخل العنوان والنص');
      return;
    }
    setState(() => _sending = true);
    try {
      if (_toAll) {
        await _admin.sendToAll(title, body);
      } else {
        if (_specialization.text.trim().isEmpty) {
          _notify('أدخل التخصص');
          setState(() => _sending = false);
          return;
        }
        await _admin.sendToSpecialization(_specialization.text.trim(), title, body);
      }
      _title.clear();
      _body.clear();
      _specialization.clear();
      if (mounted) _notify('تم إرسال الإشعار');
    } catch (_) {
      _notify('تعذّر الإرسال');
    } finally {
      if (mounted) setState(() => _sending = false);
    }
  }

  void _notify(String msg) {
    if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(msg)));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('إرسال إشعار')),
      body: ListView(
        padding: const EdgeInsets.all(20),
        children: [
          SegmentedButton<bool>(
            segments: const [
              ButtonSegment(value: true, label: Text('لجميع المستخدمين')),
              ButtonSegment(value: false, label: Text('حسب التخصص')),
            ],
            selected: {_toAll},
            onSelectionChanged: (s) => setState(() => _toAll = s.first),
          ),
          const SizedBox(height: 16),
          TextField(controller: _title, decoration: const InputDecoration(labelText: 'العنوان')),
          const SizedBox(height: 16),
          TextField(controller: _body, maxLines: 4, decoration: const InputDecoration(labelText: 'النص')),
          if (!_toAll) ...[
            const SizedBox(height: 16),
            TextField(controller: _specialization, decoration: const InputDecoration(labelText: 'التخصص (مثال: الرياضيات)')),
          ],
          const SizedBox(height: 24),
          SizedBox(
            width: double.infinity,
            child: FilledButton(
              onPressed: _sending ? null : _send,
              child: _sending
                  ? const SizedBox(width: 22, height: 22, child: CircularProgressIndicator(strokeWidth: 2.5, color: Colors.white))
                  : const Text('إرسال'),
            ),
          ),
        ],
      ),
    );
  }
}

import 'package:file_picker/file_picker.dart';
import 'package:flutter/material.dart';

import '../../services/admin_service.dart';

class AdminSettings extends StatefulWidget {
  const AdminSettings({super.key});

  @override
  State<AdminSettings> createState() => _AdminSettingsState();
}

class _AdminSettingsState extends State<AdminSettings> {
  final _admin = AdminService();
  final _appName = TextEditingController();
  final _appVersion = TextEditingController();
  final _telegram = TextEditingController();
  final _youtube = TextEditingController();
  final _facebook = TextEditingController();
  final _instagram = TextEditingController();
  bool _loading = true;
  bool _saving = false;

  @override
  void initState() {
    super.initState();
    _load();
  }

  @override
  void dispose() {
    _appName.dispose();
    _appVersion.dispose();
    _telegram.dispose();
    _youtube.dispose();
    _facebook.dispose();
    _instagram.dispose();
    super.dispose();
  }

  Future<void> _load() async {
    try {
      final s = await _admin.fetchSettings();
      if (!mounted) return;
      setState(() {
        _appName.text = s['app_name'] ?? '';
        _appVersion.text = s['app_version'] ?? '';
        _telegram.text = s['link_telegram'] ?? '';
        _youtube.text = s['link_youtube'] ?? '';
        _facebook.text = s['link_facebook'] ?? '';
        _instagram.text = s['link_instagram'] ?? '';
        _loading = false;
      });
    } catch (_) {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _save() async {
    setState(() => _saving = true);
    try {
      await _admin.updateSettings({
        'app_name': _appName.text.trim(),
        'app_version': _appVersion.text.trim(),
        'link_telegram': _telegram.text.trim(),
        'link_youtube': _youtube.text.trim(),
        'link_facebook': _facebook.text.trim(),
        'link_instagram': _instagram.text.trim(),
      });
      if (mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(const SnackBar(content: Text('تم حفظ الإعدادات')));
      }
    } catch (_) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('تعذّر الحفظ')));
      }
    } finally {
      if (mounted) setState(() => _saving = false);
    }
  }

  Future<void> _uploadLogo() async {
    final result = await FilePicker.platform.pickFiles(type: FileType.image, withData: false);
    if (result == null || result.files.isEmpty || result.files.first.path == null) return;
    try {
      await _admin.uploadLogo(result.files.first.path!);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('تم رفع الشعار')));
      }
    } catch (_) {
      if (mounted) {
        ScaffoldMessenger.of(context)
            .showSnackBar(const SnackBar(content: Text('تعذّر رفع الشعار')));
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(

  @override
  void initState() {

    _isFavorite = widget.file.isFavorite;

  }

  @override
  void dispose() {

    super.dispose();
  }

  Future<void> _load() async {
    _isOffline = await _offline.isAvailableOffline(_file.id);
    if (!mounted) return;
    await _loadComments();
    if (mounted) setState(() {});
  }

  Future<void> _loadComments() async {

      final list = await _comments.fetchComments(_file.id);
      setState(() {
        _commentList = list;
        _loadingComments = false;
      });
    } catch (_) {

    }
  }

  Future<void> _toggleFavorite() async {
    try {
      if (_isFavorite) {
        await _library.removeFavorite(_file.id);
      } else {
        await _library.addFavorite(_file.id);
      }
      setState(() => _isFavorite = !_isFavorite);
      _notify('تمت العملية');
    } catch (_) {
      _notify('حدث خطأ');
    }
  }

  Future<void> _rate(int value) async {
    final result = await _library.rate(_file.id, value);
    setState(() {
      _file = _file.copyWith(
        myRating: value,
        avgRating: (result['avg_rating'] as num?)?.toDouble() ?? _file.stats.avgRating,
      );
    });
  }

  Future<void> _read() async {
    // "قراءة" - downloads the document and opens/appends it to content.
    await _downloadFile();
  }

  Future<void> _downloadFile() async {
    try {
      final saved = await _downloadContent(_fileUrl, _file.filename);
      _notify('تم التحميل: ${saved.split('/').last}');
    } catch (_) {
      _notify('تعذّر التحميل');
    }
  }

  Future<String> _downloadContent(String url, String filename) async {
    final client = HttpClient();
    final request = await client.getUrl(Uri.parse(url));
    final token = ApiService().token;
    if (token != null && token.isNotEmpty) {
      request.headers.set(HttpHeaders.authorizationHeader, 'Bearer $token');
    }
    final response = await request.close();
    if (response.statusCode >= 400) {
      throw Exception('Download failed');
    }
    final bytes = await _collectBytes(response);
    client.close();

    final base = await getApplicationDocumentsDirectory();
    final folder = Directory('${base.path}/asatezah_downloads');
    if (!await folder.exists()) await folder.create(recursive: true);
    final file = File('${folder.path}/$filename');
    await file.writeAsBytes(bytes, flush: true);
    return file.path;
  }

  Future<Uint8List> _collectBytes(HttpClientResponse response) async {
    final builder = BytesBuilder();
    await for (final chunk in response) {
      builder.add(chunk);
    }
    return builder.takeBytes();
  }

  Future<void> _toggleOffline() async {
    try {
      if (_isOffline) {
        await _offline.removeOffline(_file.id);
        setState(() => _isOffline = false);
        _notify('أُزيل من دون اتصال');

  Future<void> _loadMore({bool reset = false}) async {
    if (_loading) return;
    if (!reset && !_hasMore) return;
    setState(() => _loading = true);
    final page = reset ? 1 : _page + 1;
    try {
      final result = await _library.fetchFiles(
        page: page,
        sort: _sort,
        categoryId: _categoryId,
        materialId: _materialId,
        minRating: _minRating,
      );
      setState(() {
        if (reset) {
          _all = result.files;
          _page = 1;
        } else {
          _all = [..._all, ...result.files];
          _page = page;
        }
        _hasMore = page < result.lastPage;
      });
    } catch (e) {
      if (mounted && reset) {
        ScaffoldMessenger.of(context)
            .showSnackBar(SnackBar(content: Text('تعذّر تحميل الملفات: $e')));
      }
    } finally {
      if (mounted) setState(() => _loading = false);

  Future<void> _loadCatalog() async {
    try {
      final filters = await _library.fetchFilters();
      if (!mounted) return;
      setState(() {
        _categories = filters.categories;
        _materials = filters.materials;
        _loadingCatalog = false;
      });
    } catch (_) {
      if (mounted) setState(() => _loadingCatalog = false);
    }
  }

  Future<void> _pickFile() async {
    final result = await FilePicker.platform.pickFiles(withData: false);
    if (result != null && result.files.isNotEmpty) {
      final f = result.files.first;
      setState(() {
        _filePath = f.path;
        _fileName = f.name;
      });
    }
  }

  Future<void> _submit() async {
    if (_filePath == null) {
      _notify('اختر ملفاً أولاً');
      return;
    }
    final title = _titleController.text.trim();
    if (title.isEmpty) {
      _notify('أدخل اسم الملف');
      return;
    }
    setState(() => _uploading = true);
    try {
      await _upload.upload(
        filePath: _filePath!,
        title: title,
        description: _descController.text.trim().isEmpty ? null : _descController.text.trim(),
        categoryId: _categoryId,
        materialId: _materialId,
      );
      if (!mounted) return;
      Navigator.pop(context);
      ScaffoldMessenger.of(context)
          .showSnackBar(const SnackBar(content: Text('تم إرسال الملف للمراجعة')));
    } catch (_) {
      // Duplicate content is rejected silently by the backend; show nothing
      // specific, just acknowledge the submission attempt without error.
      if (mounted) setState(() => _uploading = false);
    }
  }

  void _notify(String msg) {
    if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(msg)));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('رفع ملف')),
      body: SingleChildScrollView(
        padding: const EdgeInsets.all(20),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            _filePickerCard(),
            const SizedBox(height: 20),
            TextField(
              controller: _titleController,
              decoration: const InputDecoration(
                labelText: 'اسم الملف *',
                hintText: 'مثال: درس الرياضيات - الوحدة الأولى',
              ),
            ),
            const SizedBox(height: 16),
            TextField(
              controller: _descController,
              maxLines: 3,
              decoration: const InputDecoration(labelText: 'الوصف (اختياري)'),
            ),
            if (_loadingCatalog)
              const Padding(
                padding: EdgeInsets.all(16),
                child: Center(child: CircularProgressIndicator()),
              )
            else ...[
              const SizedBox(height: 20),
              Text('التخصص / الصف', style: Theme.of(context).textTheme.titleSmall),
              const SizedBox(height: 8),
              _catalogDropdown(
                _categories,
                (v) => _categoryId = v,
                _categoryId,
                hint: 'اختر التصنيف',
              ),
              const SizedBox(height: 16),
              Text('المادة', style: Theme.of(context).textTheme.titleSmall),
              const SizedBox(height: 8),
              _catalogDropdown(
                _materials,
                (v) => _materialId = v,
                _materialId,
                hint: 'اختر المادة',
              ),
            ],
            const SizedBox(height: 28),
            SizedBox(
              width: double.infinity,
              child: FilledButton(
                onPressed: _uploading ? null : _submit,
                child: _uploading
                    ? const SizedBox(
                        width: 22,
                        height: 22,
                        child: CircularProgressIndicator(strokeWidth: 2.5, color: Colors.white),
                      )
                    : const Text('إرسال للمراجعة'),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _catalogDropdown(
    List<CatalogItem> items,
    ValueChanged<int?> onChanged,
    int? value, {
    required String hint,
  }) {
    return DropdownButtonFormField<int>(
      initialValue: value,
      hint: Text(hint),
      isExpanded: true,
      items: items.map((i) => DropdownMenuItem(value: i.id, child: Text(i.name))).toList(),
      onChanged: (v) => setState(() => onChanged(v)),
      decoration: const InputDecoration(border: OutlineInputBorder()),
    );
  }

  Widget _filePickerCard() {
    return InkWell(
      onTap: _pickFile,
      borderRadius: BorderRadius.circular(16),
      child: Container(
        height: 140,
        width: double.infinity,
        decoration: BoxDecoration(
          border: Border.all(color: const Color(0xFFE5E7EB), width: 1.5),
          borderRadius: BorderRadius.circular(16),
        ),
        child: _filePath == null
            ? Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: const [
                  Icon(Icons.cloud_upload_outlined, size: 48, color: Colors.grey),
                  SizedBox(height: 8),
                  Text('اختر ملفاً للرفع', style: TextStyle(color: Colors.grey)),
                  SizedBox(height: 4),
                  Text(
                    'PDF, Word, Excel, PowerPoint, صور، فيديو',
                    style: TextStyle(fontSize: 11, color: Colors.grey),
                  ),
                ],
              )
            : Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  FileTypeIcon(filename: _fileName ?? '', size: 48),
                  const SizedBox(height: 8),
                  Text(
                    _fileName ?? '',
                    textAlign: TextAlign.center,
                    style: const TextStyle(fontWeight: FontWeight.bold),
                  ),
                  const SizedBox(height: 4),
                  const Text(
                    'اضغط لتغيير الملف',
                    style: TextStyle(fontSize: 11, color: Colors.grey),
                  ),
                ],
              ),
      ),
    );
  }
